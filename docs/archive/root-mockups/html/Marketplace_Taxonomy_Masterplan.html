📐 1. Existierender Blueprint — Was funktioniert

Kaufland läuft in Production ohne manuelles Mapping. Das ist der Beweis, dass der KI-First-Ansatz funktioniert. Hier die exakte Architektur, die wir für alle neuen Marktplätze replizieren.

// scripts/fetch-amazon-taxonomy.js

const SellingPartnerAPI = require('amazon-sp-api');

async function fetchBrowseTree() {
  const sp = new SellingPartnerAPI({ region: 'eu', ... });

  // 1. Report anfordern
  const report = await sp.callAPI({
    operation: 'createReport',
    body: {
      reportType: 'GET_XML_BROWSE_TREE_DATA',
      marketplaceIds: ['A1PA6795UKMFR9']  // DE
    }
  });

  // 2. Warten bis fertig (polling)
  let status;
  do {
    await sleep(30000);
    status = await sp.callAPI({
      operation: 'getReport',
      path: { reportId: report.reportId }
    });
  } while (status.processingStatus !== 'DONE');

  // 3. Report downloaden (XML)
  const doc = await sp.callAPI({
    operation: 'getReportDocument',
    path: { reportDocumentId: status.reportDocumentId }
  });

  // 4. XML → CSV konvertieren
  const xml = await downloadAndDecompress(doc.url);
  const categories = parseAmazonBrowseTree(xml);

  // 5. Speichern als CSV
  writeCsv('taxonomy-data/amazon/browse-tree-de.csv', categories);
  // Spalten: id, name, path, level, parentId, isLeaf
}

// Für jede relevante Kategorie: Pflichtfelder abfragen

async function fetchProductTypeSchemas() {
  const sp = new SellingPartnerAPI({ region: 'eu', ... });

  // 1. Alle Product Types listen
  const types = await sp.callAPI({
    operation: 'searchDefinitionsProductTypes',
    query: { marketplaceIds: ['A1PA6795UKMFR9'] }
  });

  // 2. Pro Type das Schema holen
  for (const type of types.productTypes) {
    const schema = await sp.callAPI({
      operation: 'getDefinitionsProductType',
      path: { productType: type.name },
      query: {
        marketplaceIds: ['A1PA6795UKMFR9'],
        requirements: 'LISTING',
        locale: 'de_DE'
      }
    });

    // 3. Schema → Pflichtfelder extrahieren
    const required = extractRequiredFields(schema.schema);

    fs.writeFileSync(
      `taxonomy-data/amazon/type-schemas/${type.name}.json`,
      JSON.stringify({ type: type.name, required, schema: schema.schema })
    );

    await sleep(200); // Rate Limit respektieren
  }
}

Otto — Paginiertes Flat-List API

// scripts/fetch-otto-taxonomy.js

async function fetchOttoCategories() {
  const token = await getOttoAccessToken();
  let allCategories = [];
  let page = 0;
  let hasMore = true;

  while (hasMore) {
    const response = await fetch(
      `https://api.otto.market/v3/products/categories?limit=100&page=${page}`,
      { headers: { Authorization: `Bearer ${token}` } }
    );
    const data = await response.json();

    allCategories.push(...data.categories);
    hasMore = data.categories.length === 100;
    page++;

    await sleep(100); // 20 req/s Limit
  }

  // Normalisieren → Unified Format
  const normalized = allCategories.map(cat => ({
    id: cat.categoryGroupId,
    name: cat.categoryGroupName,
    path: cat.categoryGroupName, // Otto hat keine Hierarchie-Pfade
    attributes: cat.requiredAttributes || [],
    marketplace: 'otto'
  }));

  fs.writeFileSync(
    'taxonomy-data/otto/categories.json',
    JSON.stringify(normalized, null, 2)
  );

  console.log(`Otto: ${normalized.length} Kategorien gespeichert`);
}
Otto Besonderheit: Otto hat eine flache Kategorie-Struktur ohne tiefe Hierarchie. Der Kategorie-Name muss EXAKT zur API-Erwartung passen. Token-Scoring muss hier besonders präzise sein. Empfehlung: Gemini-Fallback als primären Matcher nutzen (weniger Kategorien = besser für LLM).

Etsy — Hierarchical Tree + Properties

// scripts/fetch-etsy-taxonomy.js

async function fetchEtsyTaxonomy() {
  // 1. Ganzen Seller Taxonomy Tree holen (1 Call!)
  const tree = await fetch(
    'https://openapi.etsy.com/v3/application/seller-taxonomy/nodes',
    { headers: { 'x-api-key': ETSY_API_KEY } }
  );
  const nodes = await tree.json();

  // 2. Flattening: Tree → Array mit Pfaden
  const flat = flattenTree(nodes.results, '');

  fs.writeFileSync(
    'taxonomy-data/etsy/seller-taxonomy.json',
    JSON.stringify(flat, null, 2)
  );

  // 3. Für jede Leaf-Kategorie: Properties abfragen
  const leafNodes = flat.filter(n => n.isLeaf);
  const properties = {};

  for (const node of leafNodes) {
    const props = await fetch(
      `https://openapi.etsy.com/v3/application/seller-taxonomy/nodes/${node.id}/properties`,
      { headers: { 'x-api-key': ETSY_API_KEY } }
    );
    properties[node.id] = await props.json();

    await sleep(110); // 10 QPS Limit
    // ~10.000 Leaf-Nodes × 0.11s = ~18 Minuten
  }

  fs.writeFileSync(
    'taxonomy-data/etsy/properties.json',
    JSON.stringify(properties, null, 2)
  );
}

function flattenTree(nodes, parentPath) {
  const result = [];
  for (const node of nodes) {
    const path = parentPath
      ? `${parentPath} > ${node.name}`
      : node.name;

    result.push({
      id: node.id,
      name: node.name,
      path,
      normalized: path.toLowerCase(),
      level: node.level,
      parentId: node.parent_id,
      isLeaf: !node.children || node.children.length === 0,
      marketplace: 'etsy'
    });

    if (node.children) {
      result.push(...flattenTree(node.children, path));
    }
  }
  return result;
}

async function fetchShopifyTaxonomy() {

WooCommerce / Shopware / Magento — Shop-spezifisch

Fundamentaler Unterschied zu Marktplätzen: Shops haben KEINEN globalen Kategorie-Baum. Jeder Shop hat seine eigenen Kategorien. Die Taxonomie-Daten werden NICHT vorinstalliert, sondern bei jedem Connect frisch aus der Shop-API gezogen. 
// lib/shop-taxonomy-sync.js — Dynamisches Category Fetching

async function syncShopCategories(integration) {
  const { type, tenantId, credentials } = integration;

  let categories;
  switch (type) {
    case 'woocommerce':
      categories = await fetchWooCommerceCategories(credentials);
      break;
    case 'shopware':
      categories = await fetchShopwareCategories(credentials);
      break;
    case 'magento':
      categories = await fetchMagentoCategories(credentials);
      break;
  }

  // In Firestore speichern (pro Tenant + Shop)
  await getDb()
    .collection('shop_taxonomies')
    .doc(`${tenantId}__${type}`)
    .set({
      categories: categories.map(c => ({
        id: c.id,
        name: c.name,
        path: c.path,
        normalized: c.path.toLowerCase(),
        parentId: c.parentId,
        productCount: c.count
      })),
      lastSynced: new Date().toISOString(),
      categoryCount: categories.length
    });

  return categories;
}

async function fetchWooCommerceCategories({ url, key, secret }) {
  const WooCommerce = require('@woocommerce/woocommerce-rest-api');
  const api = new WooCommerce.default({ url, consumerKey: key, consumerSecret: secret });

  let all = [], page = 1;
  while (true) {
    const { data } = await api.get('products/categories', { per_page: 100, page });
    if (!data.length) break;
    all.push(...data);
    page++;
  }

  // Pfade aus Parent-Beziehungen aufbauen
  return buildPathsFromParents(all);
}

// services/integration-ai-config.js — Shop Category Mapping

async function autoMapShopCategories(tenantId, shopType) {
  // 1. Produkte des Users laden

  const shopCats = await getShopTaxonomy(tenantId, shopType);

  const prompt = `Du bist ein E-Commerce Experte.

🤖 5. KI-Engine — Intelligentes Category Matching

Das Herzstück des KI-First-Ansatzes: Gemini wählt aus einer vorselektierten Shortlist die beste Kategorie. Kein Halluzinieren möglich, weil nur echte Kategorie-IDs als Antwort akzeptiert werden.

Gemini Category Resolution — Universell

// lib/category-matcher.js — Universelle 4-Tier Resolution

const { loadTaxonomy } = require('./taxonomy-loader');
const { geminiStructured } = require('./gemini-structured');

// LRU Cache um doppelte Gemini-Calls zu vermeiden
const LRU = require('lru-cache');
const cache = new LRU({ max: 10000, ttl: 1000 * 60 * 60 * 24 }); // 24h

/**
 * Universelle Kategorie-Resolution für JEDEN Marktplatz.
 * Repliziert das bewährte Kaufland/eBay-Pattern.
 */
async function resolveCategory(marketplace, product) {
  const cacheKey = `${marketplace}:${product.id || product.title}`;
  const cached = cache.get(cacheKey);
  if (cached) return cached;

  const taxonomy = loadTaxonomy(marketplace);
  if (!taxonomy || !taxonomy.categories.length) return null;

  const details = product.details || {};
  const attributes = details.attributes || {};
  const identification = product.identification || {};

  // ── TIER 1: Direct ID Validation ──
  const rawId = attributes[`${marketplace}_category_id`]
             || details[`${marketplace}CategoryId`];

  if (rawId && taxonomy.isValidId(rawId)) {
    const cat = taxonomy.getById(rawId);
    if (cat) {
      const result = { id: rawId, path: cat.path, tier: 1, confidence: 1.0 };
      cache.set(cacheKey, result);
      return result;
    }
  }

  // ── TIER 2: Path Lookup ──
  const rawPath = details[`${marketplace}CategoryPath`]
               || attributes[`${marketplace}_category`]
               || identification.category;

  if (rawPath) {
    const byPath = taxonomy.lookupByPath(rawPath);
    if (byPath) {
      const result = { id: byPath.id, path: byPath.path, tier: 2, confidence: 0.95 };
      cache.set(cacheKey, result);
      return result;
    }
  }

  // ── TIER 3: Token Scoring (Fuzzy Match) ──
  const searchText = [
    product.title,
    rawPath,
    identification.category,
    details.category,
    attributes.Kategorie
  ].filter(Boolean).join(' ');

  if (searchText) {
    const scored = taxonomy.fuzzyMatch(searchText);
    if (scored && scored.score > 8) { // Mindest-Score
      const result = { id: scored.id, path: scored.path, tier: 3, confidence: scored.score / 20 };
      cache.set(cacheKey, result);
      return result;
    }
  }

  // ── TIER 4: Gemini AI ──
  const shortlist = taxonomy.buildShortlist(product, 60);
  if (!shortlist.length) return null;

  const geminiResult = await resolveWithGemini(marketplace, product, shortlist);
  if (geminiResult) {
    const result = { ...geminiResult, tier: 4 };
    cache.set(cacheKey, result);
    return result;
  }

  return null;
}

/**
 * Gemini wählt aus Shortlist — KEINE Halluzination möglich.
 */
async function resolveWithGemini(marketplace, product, shortlist) {
  const productInfo = [
    product.title,
    product.details?.brand,
    product.details?.description?.substring(0, 200),
    product.identification?.category
  ].filter(Boolean).join(' | ');

  const candidateList = shortlist
    .map(c => `ID:${c.id} → ${c.path}`)
    .join('\n');

  const prompt = `Du bist ein Kategorisierungs-Assistent für ${marketplace}.
Aufgabe: Wähle GENAU EINE Kategorie aus der Kandidatenliste für dieses Produkt.

PRODUKT: ${productInfo}

KANDIDATEN:
${candidateList}

Regeln:
- Du darfst NUR eine ID aus der Kandidatenliste zurückgeben
- Wähle die spezifischste Leaf-Kategorie die passt
- Bei Unsicherheit: generellere Kategorie wählen

Antwort als JSON: {"id": "...", "confidence": 0.0-1.0}`;

  try {
    const response = await geminiStructured(prompt, {
      schema: { type: 'object', properties: { id: { type: 'string' }, confidence: { type: 'number' } } }
    });

    // Validierung: ID muss in Shortlist sein!
    const valid = shortlist.find(c => String(c.id) === String(response.id));
    if (!valid) return null;

    return { id: response.id, path: valid.path, confidence: response.confidence };
  } catch (err) {
    console.error(`[category-matcher] Gemini failed for ${marketplace}:`, err.message);
    return null;
  }
}

module.exports = { resolveCategory };

Taxonomy Loader — Unified Interface

// lib/taxonomy-loader.js

const fs = require('fs');
const path = require('path');
const csvParse = require('csv-parse/sync');

const TAXONOMY_DIR = path.join(__dirname, '..', 'taxonomy-data');
const taxonomyCache = new Map();

/**
 * Lädt Taxonomie für einen Marktplatz.
 * Gibt ein TaxonomyIndex-Objekt zurück mit einheitlicher API.
 */
function loadTaxonomy(marketplace) {
  if (taxonomyCache.has(marketplace)) return taxonomyCache.get(marketplace);

  const configPath = path.join(TAXONOMY_DIR, marketplace, 'config.json');
  if (!fs.existsSync(configPath)) {
    console.warn(`[taxonomy-loader] No config for ${marketplace}`);
    return null;
  }

  const config = JSON.parse(fs.readFileSync(configPath, 'utf8'));
  let categories;

  switch (config.format) {
    case 'csv':
      categories = loadFromCsv(marketplace, config);
      break;
    case 'json':
      categories = loadFromJson(marketplace, config);
      break;
    default:
      throw new Error(`Unknown format: ${config.format}`);
  }

  const index = new TaxonomyIndex(marketplace, categories);
  taxonomyCache.set(marketplace, index);

  console.log(`[taxonomy-loader] ${marketplace}: ${categories.length} categories loaded`);
  return index;
}

class TaxonomyIndex {
  constructor(marketplace, categories) {
    this.marketplace = marketplace;
    this.categories = categories;

    // Build indices
    this.idMap = new Map();
    this.pathIndex = new Map();
    this.nameIndex = new Map();
    this.idSet = new Set();

    const nameCounts = new Map();

    for (const cat of categories) {
      this.idMap.set(String(cat.id), cat);
      this.idSet.add(String(cat.id));
      this.pathIndex.set(cat.normalized, cat);

      // Unique leaf names only
      const leafName = cat.name?.toLowerCase().trim();
      if (leafName) {
        nameCounts.set(leafName, (nameCounts.get(leafName) || 0) + 1);
      }
    }

    for (const cat of categories) {
      const leafName = cat.name?.toLowerCase().trim();
      if (leafName && nameCounts.get(leafName) === 1) {
        this.nameIndex.set(leafName, cat);
      }
    }
  }

  isValidId(id) { return this.idSet.has(String(id).trim()); }
  getById(id) { return this.idMap.get(String(id).trim()); }

  lookupByPath(pathOrName) {
    const norm = pathOrName.toString().trim().toLowerCase();
    return this.pathIndex.get(norm) || this.nameIndex.get(norm) || null;
  }

  fuzzyMatch(searchText) {
    const needle = searchText.toLowerCase().trim();
    const tokens = needle.split(/[\s\/,;>\-]+/).filter(t => t.length > 2);

    let best = null, bestScore = 0;

    for (const cat of this.categories) {
      let score = 0;

      // Substring match bonus
      if (cat.normalized.includes(needle)) score += needle.length + 5;
      else if (needle.includes(cat.normalized)) score += cat.normalized.length;

      // Token matching
      for (const token of tokens) {
        if (cat.normalized.includes(token)) score += token.length;
      }

      // Leaf preference bonus
      if (cat.isLeaf) score += 2;

      if (score > bestScore) {
        bestScore = score;
        best = { ...cat, score };
      }
    }

    return best;
  }

  buildShortlist(product, maxCandidates = 60) {
    const text = [
      product.title,
      product.details?.brand,
      product.details?.description?.substring(0, 300),
      product.identification?.category,
      product.details?.category
    ].filter(Boolean).join(' ').toLowerCase();

    const tokens = text.split(/[\s\/,;>\-]+/).filter(t => t.length > 2);

    const scored = this.categories
      .map(cat => {
        let score = 0;
        for (const token of tokens) {
          if (cat.normalized.includes(token)) score += token.length;
        }
        return { ...cat, score };
      })
      .filter(c => c.score > 0)
      .sort((a, b) => b.score - a.score)
      .slice(0, maxCandidates);

    return scored;
  }
}

module.exports = { loadTaxonomy, TaxonomyIndex };
